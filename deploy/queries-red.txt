# 服务端平均耗时（毫秒）= 耗时之和的速率 ÷ 次数的速率
server_mean_ms      1000 * sum(rate(http_request_duration_seconds_sum{route="${route}"}[${window}])) / sum(rate(http_request_duration_seconds_count{route="${route}"}[${window}]))
# 在途请求数：每次抓取时的瞬时值，在窗口里取平均（Little's Law 的 L）
inflight_avg        avg_over_time(http_inflight_requests{route="${route}"}[${window}])
# 进程每秒用掉的 CPU 秒数，即平均占用几个核
cpu_cores           rate(process_cpu_seconds_total{job="enrolld"}[${window}])
# 平均每个请求用掉的 CPU（毫秒）
cpu_ms_per_req      1000 * sum(rate(process_cpu_seconds_total{job="enrolld"}[${window}])) / sum(rate(http_requests_total{route="${route}"}[${window}]))
# CPU 的饱和，两个层次：内核运行队列里平均有几个 enrolld 的线程在等核；goroutine 在 Go 运行时里等执行线程的 p99
cpu_wait_threads    rate(process_cpu_wait_seconds_total{job="enrolld"}[${window}])
sched_wait_p99_ms   1000 * histogram_quantile(0.99, rate(go_sched_latencies_seconds_bucket{job="enrolld"}[${window}]))
# 连接池：平均占用、等连接的 p99、每次查询实际占用连接的时间
pool_in_use_avg     avg_over_time(db_pool_in_use[${window}])
pool_wait_p99_ms    1000 * histogram_quantile(0.99, sum by (le) (rate(db_pool_wait_seconds_bucket[${window}])))
db_query_mean_ms    1000 * rate(db_query_duration_seconds_sum[${window}]) / rate(db_query_duration_seconds_count[${window}])
